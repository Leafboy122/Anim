--// FUNCIONES JSON (sin deps)
local function JSONEncode(t)
	local result = {}
	local function encode(val)
		local t = type(val)
		if t == "string" then
			return '"' .. val:gsub('"', '\\"'):gsub("\n", "\\n") .. '"'
		elseif t == "number" then
			return tostring(val)
		elseif t == "boolean" then
			return val and "true" or "false"
		elseif t == "table" then
			local isArray = #val > 0
			local parts = {}
			if isArray then
				for i = 1, #val do
					table.insert(parts, encode(val[i]))
				end
				return "[" .. table.concat(parts, ",") .. "]"
			else
				for k, v in pairs(val) do
					table.insert(parts, '"' .. k .. '":' .. encode(v))
				end
				return "{" .. table.concat(parts, ",") .. "}"
			end
		else
			return "null"
		end
	end
	return encode(t)
end

local function JSONDecode(str)
	local pos = 1
	local function skipWs()
		while pos <= #str and str:sub(pos, pos):match("%s") do
			pos = pos + 1
		end
	end
	local function parseValue()
		skipWs()
		local c = str:sub(pos, pos)
		if c == '"' then
			pos = pos + 1
			local start = pos
			while pos <= #str do
				if str:sub(pos, pos) == '"' then
					local val = str:sub(start, pos - 1)
					pos = pos + 1
					return val:gsub('\\"', '"'):gsub("\\n", "\n")
				end
				pos = pos + 1
			end
		elseif c == "{" then
			pos = pos + 1
			local obj = {}
			skipWs()
			if str:sub(pos, pos) ~= "}" then
				while true do
					skipWs()
					local key = parseValue()
					skipWs()
					pos = pos + 1
					skipWs()
					obj[key] = parseValue()
					skipWs()
					if str:sub(pos, pos) == "}" then break end
					pos = pos + 1
				end
			end
			pos = pos + 1
			return obj
		elseif c == "[" then
			pos = pos + 1
			local arr = {}
			skipWs()
			if str:sub(pos, pos) ~= "]" then
				while true do
					table.insert(arr, parseValue())
					skipWs()
					if str:sub(pos, pos) == "]" then break end
					pos = pos + 1
					skipWs()
				end
			end
			pos = pos + 1
			return arr
		elseif c == "t" then
			pos = pos + 4
			return true
		elseif c == "f" then
			pos = pos + 5
			return false
		elseif c == "n" then
			pos = pos + 4
			return nil
		else
			local start = pos
			while pos <= #str and not str:sub(pos, pos):match("[,}%]%s]") do
				pos = pos + 1
			end
			local numStr = str:sub(start, pos - 1)
			return tonumber(numStr) or 0
		end
	end
	return parseValue()
end

--// CARGAR CONFIG DEL DISCO
local function loadConfig()
	if not (readfile and isfile) then
		warn("readfile/writefile no soportado en este executor")
		return nil
	end

	if isfile(CONFIG_FILE) then
		local success, data = pcall(function()
			return JSONDecode(readfile(CONFIG_FILE))
		end)
		if success and data then
			return data
		end
	end
	return nil
end

--// GUARDAR CONFIG AL DISCO
local function saveConfig(config)
	if not writefile then
		warn("writefile no soportado en este executor")
		return false
	end

	local success = pcall(function()
		writefile(CONFIG_FILE, JSONEncode(config))
	end)
	return success
end

--// APPLY ANIMATION (full set)
local function applyAnimations()
	local char = LocalPlayer.Character
	if not char then return end

	local animate = char:FindFirstChild("Animate")
	if not animate then return end

	local animationMap = {}

	for _, selected in pairs(selectedAnimations) do
		if selected.type == "full" then
			local pack = PACKS[selected.pack]
			if pack then
				for key, animId in pairs(pack) do
					animationMap[key] = animId
				end
			end
		end
	end

	pcall(function()
		if animationMap["WalkAnim"] or animationMap["Walk"] then
			animate.walk.WalkAnim.AnimationId = "rbxassetid://"..(animationMap["WalkAnim"] or animationMap["Walk"])
		end
		if animationMap["RunAnim"] or animationMap["Run"] then
			animate.run.RunAnim.AnimationId = "rbxassetid://"..(animationMap["RunAnim"] or animationMap["Run"])
		end
		if animationMap["JumpAnim"] or animationMap["Jump"] then
			animate.jump.JumpAnim.AnimationId = "rbxassetid://"..(animationMap["JumpAnim"] or animationMap["Jump"])
		end
		if animationMap["FallAnim"] or animationMap["Fall"] then
			animate.fall.FallAnim.AnimationId = "rbxassetid://"..(animationMap["FallAnim"] or animationMap["Fall"])
		end
		if animationMap["ClimbAnim"] or animationMap["Climb"] then
			animate.climb.ClimbAnim.AnimationId = "rbxassetid://"..(animationMap["ClimbAnim"] or animationMap["Climb"])
		end
		if animationMap["SwimIdle"] then
			animate.swimidle.SwimIdle.AnimationId = "rbxassetid://"..animationMap["SwimIdle"]
		end
		if animationMap["Swim"] then
			animate.swim.Swim.AnimationId = "rbxassetid://"..animationMap["Swim"]
		end
		if animationMap["Animation1"] then
			animate.idle.Animation1.AnimationId = "rbxassetid://"..animationMap["Animation1"]
		end
		if animationMap["Animation2"] then
			animate.idle.Animation2.AnimationId = "rbxassetid://"..animationMap["Animation2"]
		end
	end)
end

--// UPDATE LIST (todos los packs como "full")
function updateAnimationList()
	for _, child in pairs(scroll:GetChildren()) do
		if child:IsA("TextButton") then
			child:Destroy()
		end
	end

	for packName, _ in pairs(PACKS) do
		local btn = Instance.new("TextButton")
		btn.Size = UDim2.new(1, 0, 0, 28)
		btn.BackgroundColor3 = COLORS.Second
		btn.BackgroundTransparency = 0.5
		btn.Text = packName
		btn.TextColor3 = COLORS.Text
		btn.Font = Enum.Font.GothamBold
		btn.TextSize = 10
		btn.BorderSizePixel = 0
		btn.AutoButtonColor = false
		btn.Parent = scroll

		Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)
		animStroke(btn, 1)

		local function updateState()
			local isSelected = false
			for _, sel in pairs(selectedAnimations) do
				if sel.pack == packName and sel.type == "full" then
					isSelected = true
					break
				end
			end
			TweenService:Create(btn, TweenInfo.new(0.15), {
				BackgroundColor3 = isSelected and COLORS.Selected or COLORS.Second,
				BackgroundTransparency = isSelected and 0.3 or 0.5
			}):Play()
			return isSelected
		end

		btn.MouseButton1Click:Connect(function()
			local isSelected = updateState()
			
			if isSelected then
				-- Quitar selección
				for i, sel in pairs(selectedAnimations) do
					if sel.pack == packName and sel.type == "full" then
						table.remove(selectedAnimations, i)
						break
					end
				end
			else
				-- Agregar selección
				table.insert(selectedAnimations, {
					pack = packName,
					type = "full"
				})
			end

			getgenv().RobiConfig.SelectedAnimations = selectedAnimations
			saveConfig(getgenv().RobiConfig)
			updateState()
			applyAnimations()
		end)

		updateState()
	end

	layout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
		scroll.CanvasSize = UDim2.new(0, 0, 0, layout.AbsoluteContentSize.Y + 4)
	end)
end

--// DRAG
local dragging = false
local dragStart
local startPos

header.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = frame.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
				getgenv().RobiConfig.Position = {
					X = frame.Position.X.Offset,
					Y = frame.Position.Y.Offset
				}
				saveConfig(getgenv().RobiConfig)
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		local vp = workspace.CurrentCamera.ViewportSize
		local gs = frame.AbsoluteSize
		
		local newX = math.clamp(startPos.X.Offset + delta.X, -gs.X + 50, vp.X - 50)
		local newY = math.clamp(startPos.Y.Offset + delta.Y, -gs.Y + 20, vp.Y - 20)
		
		frame.Position = UDim2.new(0, newX, 0, newY)
	end
end)
